Opsi 1: Menggunakan PowerShell (Windows)
# Navigasi ke folder proyek cd "path\to\bernaunginv" # 1. Copy file gambar ke /assets/images Copy-Item -Path "images\*" -Destination "assets\images" -Force -Recurse # 3. Hapus folder lama Remove-Item -Path "images" -Force -Recurse Remove-Item -Path "tema\tema-jawa\index.html" -Force # Hapus redirect Remove-Item -Path "tema\tema-jawa2\index.html" -Force Remove-Item -Path "tema\tema-netflix\index.html" -Force Remove-Item -Path "tema\tema-spotify\index.html" -Force Remove-Item -Path "tema\tema-jawanopict\index.html" -Force Remove-Item -Path "tema\tema-deluxe\index.html" -Force Remove-Item -Path "tema\tema-simple\index.html" -Force Remove-Item -Path "tema\latif-erna\index.html" -Force Remove-Item -Path "tema\pengecekan1\index.html" -Force Remove-Item -Path "tema\Tes Tanpafoto\index.html" -Force
